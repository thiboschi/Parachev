function main(workbook: ExcelScript.Workbook) {
    const ws = workbook.getActiveWorksheet();
    supprimerEntreMots(ws, ["KO"], ["EN"]);
    convertirEnTableau(ws, ["BO", "SI", "AK", "IK", "PU", "KO", "KA"], "EN");
    convert(ws);
    nettoyerValeurs(ws);
    convertirEnNombre(ws);
    trierColonnes(ws, ["BO", "SI", "AK", "IK", "PU", "KO", "KA"], "EN");
    ajouterBordures(ws, ["BO", "SI", "AK", "IK", "PU", "KO", "KA"], "EN");
    alignerNombres(ws);
    formaterDecimales(ws);
}

function supprimerEntreMots(ws: ExcelScript.Worksheet, motsDebut: string[], motsFin: string[]) {
    const ligneDepart = 22;
    const colonneRecherche = 0;
    const inclureBornes = false;
    const usedRange = ws.getUsedRange();
    const valeurs = usedRange.getValues();
    const nbLignes = valeurs.length;
    let lignesASupprimer: number[] = [];
    let recherche = true;
    let depart = ligneDepart;
    while (recherche) {
        let ligneDebut = -1;
        let ligneFin = -1;
        for (let i = depart; i < nbLignes; i++) {
            const cellule = String(valeurs[i][colonneRecherche]);
            for (const mot of motsDebut) {
                if (cellule.toLowerCase().includes(mot.toLowerCase())) { ligneDebut = i; break; }
            }
            if (ligneDebut !== -1) break;
        }
        if (ligneDebut === -1) { recherche = false; break; }
        for (let i = ligneDebut + 1; i < nbLignes; i++) {
            const cellule = String(valeurs[i][colonneRecherche]);
            for (const mot of motsFin) {
                if (cellule.toLowerCase().includes(mot.toLowerCase())) { ligneFin = i; break; }
            }
            if (ligneFin !== -1) break;
        }
        if (ligneFin === -1) { recherche = false; break; }
        const debutSuppr = inclureBornes ? ligneDebut : ligneDebut + 1;
        const finSuppr = inclureBornes ? ligneFin : ligneFin - 1;
        for (let i = debutSuppr; i <= finSuppr; i++) { lignesASupprimer.push(i); }
        depart = ligneFin + 1;
    }
    lignesASupprimer.sort((a, b) => b - a);
    for (const ligne of lignesASupprimer) {
        ws.getRangeByIndexes(ligne, 0, 1, usedRange.getColumnCount()).delete(ExcelScript.DeleteShiftDirection.up);
    }
    console.log(`${lignesASupprimer.length} ligne(s) supprimée(s).`);
}

function convertirEnTableau(ws: ExcelScript.Worksheet, motsDebut: string[], motFin: string) {
    const ligneDepart = 22;
    const colonneRecherche = 0;
    const inclureBornes = false;
    const delimiteur = " ";
    const usedRange = ws.getUsedRange();
    const valeurs = usedRange.getValues();
    const nbLignes = valeurs.length;
    let depart = ligneDepart;
    let recherche = true;
    while (recherche) {
        let ligneDebut = -1;
        let ligneFin = -1;
        let motDebutTrouve = "";
        for (let i = depart; i < nbLignes; i++) {
            const cellule = String(valeurs[i][colonneRecherche]);
            for (const mot of motsDebut) {
                if (cellule.toLowerCase().includes(mot.toLowerCase())) { ligneDebut = i; motDebutTrouve = mot; break; }
            }
            if (ligneDebut !== -1) break;
        }
        if (ligneDebut === -1) { recherche = false; break; }
        for (let i = ligneDebut + 1; i < nbLignes; i++) {
            const cellule = String(valeurs[i][colonneRecherche]);
            if (cellule.toLowerCase().includes(motFin.toLowerCase())) { ligneFin = i; break; }
        }
        if (ligneFin === -1) { recherche = false; break; }
        const debutConversion = inclureBornes ? ligneDebut : ligneDebut + 1;
        const finConversion = inclureBornes ? ligneFin : ligneFin - 1;
        console.log(`Conversion entre "${motDebutTrouve}" (ligne ${ligneDebut + 1}) et "${motFin}" (ligne ${ligneFin + 1})`);
        for (let i = debutConversion; i <= finConversion; i++) {
            const texte = String(valeurs[i][colonneRecherche]);
            let morceaux = texte.split(delimiteur).map(m => m.trim()).filter(m => m !== "");
            if (morceaux.length > 0 && !/^[a-zA-Z]/.test(morceaux[0])) { morceaux = ["", ...morceaux]; }
            for (let j = 0; j < morceaux.length; j++) { ws.getCell(i, colonneRecherche + j).setValue(morceaux[j]); }
        }
        depart = ligneFin + 1;
    }
    console.log("Conversion terminée !");
}

function convert(ws: ExcelScript.Worksheet) {
    const ligneDepart = 22;
    const usedRange = ws.getUsedRange();
    const valeurs = usedRange.getValues();
    const nbLignes = valeurs.length;
    const nbColonnes = valeurs[0].length;
    let modifie = false;
    for (let i = ligneDepart; i < nbLignes; i++) {
        for (let j = 0; j < nbColonnes; j++) {
            const valeur = valeurs[i][j];
            if (typeof valeur === "string" && valeur.includes(".")) {
                valeurs[i][j] = valeur.replace(/\./g, ",");
                modifie = true;
            }
        }
    }
    if (modifie) { usedRange.setValues(valeurs); }
    console.log("Remplacement terminé !");
}

function nettoyerValeurs(ws: ExcelScript.Worksheet) {
    const ligneDepart = 22;
    const usedRange = ws.getUsedRange();
    const valeurs = usedRange.getValues();
    const nbLignes = valeurs.length;
    const nbColonnes = valeurs[0].length;
    let modifie = false;
    for (let i = ligneDepart; i < nbLignes; i++) {
        for (let j = 0; j < nbColonnes; j++) {
            const valeur = valeurs[i][j];
            if (typeof valeur === "string") {
                const matchLettre = valeur.match(/^(\d+[,.]?\d*)[a-zA-Z]+$/);
                if (matchLettre) { valeurs[i][j] = matchLettre[1]; modifie = true; }
                const valeurActuelle = String(valeurs[i][j]);
                const matchVirgule = valeurActuelle.match(/^(\d+[,.])\d(\d)$/);
                if (matchVirgule) { valeurs[i][j] = valeurActuelle.slice(0, -1); modifie = true; }
            }
        }
    }
    if (modifie) { usedRange.setValues(valeurs); }
    console.log("Nettoyage terminé !");
}

function convertirEnNombre(ws: ExcelScript.Worksheet) {
    const ligneDepart = 22;
    const usedRange = ws.getUsedRange();
    const valeurs = usedRange.getValues();
    const nbLignes = valeurs.length;
    const nbColonnes = valeurs[0].length;
    let modifie = false;
    for (let i = ligneDepart; i < nbLignes; i++) {
        for (let j = 0; j < nbColonnes; j++) {
            const valeur = valeurs[i][j];
            if (typeof valeur === "string" && valeur.trim() !== "") {
                const nombre = parseFloat(valeur.replace(",", "."));
                if (!isNaN(nombre)) { valeurs[i][j] = nombre; modifie = true; }
            }
        }
    }
    if (modifie) { usedRange.setValues(valeurs); }
    console.log("Conversion en nombre terminée !");
}

function trierColonnes(ws: ExcelScript.Worksheet, motsDebut: string[], motFin: string) {
    const ligneDepart = 22;
    const colonneRecherche = 0;
    const inclureBornes = false;
    const usedRange = ws.getUsedRange();
    const valeurs = usedRange.getValues();
    const nbLignes = valeurs.length;
    const nbColonnes = usedRange.getColumnCount();
    let ligneDebut = -1;
    let ligneFin = -1;
    for (let i = ligneDepart; i < nbLignes; i++) {
        const cellule = String(valeurs[i][colonneRecherche]);
        for (const mot of motsDebut) {
            if (cellule.toLowerCase().includes(mot.toLowerCase())) { ligneDebut = i; break; }
        }
        if (ligneDebut !== -1) break;
    }
    if (ligneDebut !== -1) {
        for (let i = ligneDebut + 1; i < nbLignes; i++) {
            const cellule = String(valeurs[i][colonneRecherche]);
            if (cellule.toLowerCase().includes(motFin.toLowerCase())) { ligneFin = i; break; }
        }
    }
    if (ligneDebut === -1 || ligneFin === -1) { console.log("Marqueurs introuvables, tri annulé."); return; }
    const debutTri = inclureBornes ? ligneDebut : ligneDebut + 1;
    const finTri = inclureBornes ? ligneFin : ligneFin - 1;
    if (finTri < debutTri) { console.log("Aucune ligne à trier."); return; }
    const separateurs: number[] = [ligneDebut - 1];
    for (let i = ligneDebut; i <= finTri; i++) {
        const cellule = String(valeurs[i][colonneRecherche]);
        if (motsDebut.some(m => cellule.toLowerCase().includes(m.toLowerCase()))) { separateurs.push(i); }
    }
    separateurs.push(ligneFin);
    for (let s = 0; s < separateurs.length - 1; s++) {
        const debutSection = separateurs[s] + 1;
        const finSection = separateurs[s + 1] - 1;
        if (finSection < debutSection) continue;
        const section: (string | number | boolean)[][] = [];
        for (let i = debutSection; i <= finSection; i++) { section.push(valeurs[i]); }
        section.sort((a: (string | number | boolean)[], b: (string | number | boolean)[]) => {
            const triD = comparerValeurs(a[3], b[3]);
            if (triD !== 0) return triD;
            const triA = comparerValeurs(a[0], b[0]); // A
            if (triA !== 0) return triA;
            return comparerValeurs(a[1], b[1]); // B
        });
        ws.getRangeByIndexes(debutSection, 0, section.length, nbColonnes).setValues(section);
        console.log(`Section triée D-B-C entre les lignes ${debutSection + 1} et ${finSection + 1}.`);
    }
}

function ajouterBordures(ws: ExcelScript.Worksheet, motsDebut: string[], motFin: string) {
    const ligneDepart = 22;
    const colonneRecherche = 0;
    const largeurBordure = 5;
    const usedRange = ws.getUsedRange();
    const valeurs = usedRange.getValues();
    const nbLignes = valeurs.length;
    let ligneDebut = -1;
    let ligneFin = -1;
    for (let i = ligneDepart; i < nbLignes; i++) {
        const cellule = String(valeurs[i][colonneRecherche]);
        for (const mot of motsDebut) {
            if (cellule.toLowerCase().includes(mot.toLowerCase())) { ligneDebut = i; break; }
        }
        if (ligneDebut !== -1) break;
    }
    if (ligneDebut !== -1) {
        for (let i = ligneDebut + 1; i < nbLignes; i++) {
            const cellule = String(valeurs[i][colonneRecherche]);
            if (cellule.toLowerCase().includes(motFin.toLowerCase())) { ligneFin = i; break; }
        }
    }
    if (ligneDebut === -1 || ligneFin === -1) { console.log("Marqueurs introuvables, bordures annulées."); return; }
    const estVide = (v: string | number | boolean): boolean => String(v).trim() === "";
    let valeurAPrecedente: string | number | boolean = "";
    let valeurDPrecedente: string | number | boolean = "";
    for (let i = ligneDebut + 1; i < ligneFin; i++) {
        const estTag = motsDebut.some(m => String(valeurs[i][0]).toLowerCase().includes(m.toLowerCase()));
        if (!estTag) { valeurAPrecedente = valeurs[i][0]; valeurDPrecedente = valeurs[i][3]; break; }
    }
    for (let i = ligneDebut + 2; i < ligneFin; i++) {
        const estTag = motsDebut.some(m => String(valeurs[i][0]).toLowerCase().includes(m.toLowerCase()));
        if (estTag) continue;
        const valeurA = valeurs[i][0];
        const valeurD = valeurs[i][3];
        const changementA = !estVide(valeurA) && !estVide(valeurAPrecedente) && String(valeurA) !== String(valeurAPrecedente);
        const changementD = !estVide(valeurD) && !estVide(valeurDPrecedente) && String(valeurD) !== String(valeurDPrecedente);
        if (changementA || changementD) {
            const bordure = ws.getRangeByIndexes(i, 0, 1, largeurBordure).getFormat().getRangeBorder(ExcelScript.BorderIndex.edgeTop);
            bordure.setColor("#000000");
            bordure.setStyle(changementD ? ExcelScript.BorderLineStyle.continuous : ExcelScript.BorderLineStyle.dash);
        }
        if (!estVide(valeurA)) valeurAPrecedente = valeurA;
        if (!estVide(valeurD)) valeurDPrecedente = valeurD;
    }
    console.log("Bordures ajoutées !");
}

function alignerNombres(ws: ExcelScript.Worksheet) {
    const ligneDepart = 22;
    const usedRange = ws.getUsedRange();
    const valeurs = usedRange.getValues();
    const nbLignes = valeurs.length;
    const nbColonnes = valeurs[0].length;
    for (let i = ligneDepart; i < nbLignes; i++) {
        for (let j = 0; j < nbColonnes; j++) {
            const valeur = valeurs[i][j];
            const estNumerique = typeof valeur === "number" ||
                (typeof valeur === "string" && /^\d+([,.]\d+)?$/.test(String(valeur).trim()));
            if (estNumerique) {
                ws.getRangeByIndexes(i, j, 1, 1).getFormat().setHorizontalAlignment(ExcelScript.HorizontalAlignment.right);
            }
        }
    }
    console.log("Alignement à droite terminé !");
}

function formaterDecimales(ws: ExcelScript.Worksheet) {
    const ligneDepart = 22;
    const usedRange = ws.getUsedRange();
    const valeurs = usedRange.getValues();
    const nbLignes = valeurs.length;
    const nbColonnes = usedRange.getColumnCount();
    const aVraieDecimale = (v: string | number | boolean): boolean => {
        const num = parseFloat(String(v).replace(",", "."));
        if (isNaN(num)) return false;
        return (num * 10) % 10 !== 0;
    };
    const estVide = (v: string | number | boolean): boolean => String(v).trim() === "";
    const separateurs: number[] = [ligneDepart - 1];
    let valeurDPrecedente: string | number | boolean = valeurs[ligneDepart][3];
    for (let i = ligneDepart + 1; i < nbLignes; i++) {
        const valeurD = valeurs[i][3];
        if (!estVide(valeurD) && !estVide(valeurDPrecedente) && String(valeurD) !== String(valeurDPrecedente)) {
            separateurs.push(i - 1);
            separateurs.push(i);
        }
        if (!estVide(valeurD)) valeurDPrecedente = valeurD;
    }
    separateurs.push(nbLignes - 1);
    separateurs.sort((a, b) => a - b);
    const separateursUniques = [...new Set(separateurs)];
    for (let s = 0; s < separateursUniques.length - 1; s++) {
        const debutSection = separateursUniques[s] + 1;
        const finSection = separateursUniques[s + 1];
        if (finSection < debutSection) continue;
        let sectionAVraieDecimale = false;
        for (let i = debutSection; i <= finSection; i++) {
            for (let j = 0; j < nbColonnes; j++) {
                if (aVraieDecimale(valeurs[i][j])) { sectionAVraieDecimale = true; break; }
            }
            if (sectionAVraieDecimale) break;
        }
        const plage = ws.getRangeByIndexes(debutSection, 0, finSection - debutSection + 1, nbColonnes);
        if (sectionAVraieDecimale) {
            plage.setNumberFormat("0.0");
        } else {
            plage.setNumberFormat("0");
        }
    }
    console.log("Formatage décimales terminé !");
}

function comparerValeurs(a: string | number | boolean, b: string | number | boolean): number {
    const aStr = String(a).replace(",", ".");
    const bStr = String(b).replace(",", ".");
    const aNum = parseFloat(aStr);
    const bNum = parseFloat(bStr);
    if (!isNaN(aNum) && !isNaN(bNum)) return aNum - bNum;
    return aStr.localeCompare(bStr);
}
